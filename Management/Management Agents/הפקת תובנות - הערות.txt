export interface InsightsResult {
  summary_short: string;
  summary_full: string;
  skipped?: boolean;
  skipped_reason?: string;
}

export async function generateInsights(
  userId: number,
  options: { force?: boolean } = {}
): Promise<InsightsResult> {
  // ── Idempotency check ──────────────────────────────────────────
  const existing = await pgQueryOne<{
    personal_insights_full: string | null;
    insights_pre_completion: boolean | null;
  }>(
    "SELECT personal_insights_full, insights_pre_completion FROM users WHERE id = $1",
    [userId]
  );

  if (!options.force && existing?.personal_insights_full) {
    if (existing.insights_pre_completion === false) {
      return {
        summary_short: "",
        summary_full: "",
        skipped: true,
        skipped_reason: "final_insights_exist",
      };
    }
    // insights_pre_completion = true → regenerate with full data
    console.log([generateInsights] User ${userId}: replacing pre-completion insights with final);
  }

  // ── Load user data ─────────────────────────────────────────────
  const user = await pgQueryOne<any>("SELECT * FROM users WHERE id = $1", [userId]);
  if (!user) throw new Error(User ${userId} not found);

  // ── Build transcript ───────────────────────────────────────────
  const allMessages = await pgQueryAll<any>(
    SELECT role, content, guide, created_at FROM conversation_messages
     WHERE user_id = $1 ORDER BY created_at ASC,
    [userId]
  );
  if (!allMessages.length) throw new Error(User ${userId} has no conversation data);

  const channels: Record<string, string[]> = {};
  for (const msg of allMessages) {

    interviewer: "ראיון אישיות",

    SELECT td.internal_name, td.display_name_he, ut.score
     FROM user_traits ut JOIN trait_definitions td ON ut.trait_definition_id = td.id
     WHERE ut.user_id = $1 ORDER BY td.trait_group, td.internal_name,
    [userId]
  );

  const isFemale = user.gender === "woman";

  const readerWord = isFemale ? "הקוראת" : "הקורא";
  const thirdPerson = isFemale ? "היא" : "הוא";
  const selfWord = isFemale ? "בעצמה" : "בעצמו";
  const doWord = isFemale ? "עושה" : "עושה";
  const copesWord = isFemale ? "מתמודדת" : "מתמודד";
  const learnedWord = isFemale ? "למדת" : "למדת";
  const searchWord = isFemale ? "מחפשת" : "מחפש";
  const needWord = isFemale ? "צריכה" : "צריך";

  const systemPrompt = אתה כותב תובנות אישיות עמוקות. אתה פונה ישירות אל ${readerWord} — תמיד בגוף שני.
כל משפט שאתה כותב צריך לפנות ישירות אליך: "${youWord}..." — כאילו אתה מדבר אליו/ה פנים אל פנים.

## כלל קריטי: גוף שני בלבד
- נכון: "${youWord} ${isFemale ? "נוטה" : "נוטה"} לחפש שליטה כשמשהו מאיים ${isFemale ? "עלייך" : "עליך"}"

7. מה פחות יתאים.

כרגע כתוב “עובר בין הנושאים הבאים לא חובה”, וזה משאיר למודל יותר מדי חופש — והוא בחר לכתוב כללי.

ניסוח מתוקן לחלק המרכזי בפרומפט

הייתי מחליפה את “הגישה שלך” בערך בזה:

## הגישה שלך — תובנות מבוססות, לא סיכום ולא מחמאות

אתה לא מסכם את השיחה, אבל אתה כן משתמש בפרטים מתוכה כעוגנים.
המטרה היא לכתוב תובנות שהמשתמש/ת ירגיש/תרגיש: "וואו, באמת הבינו אותי".

אל תכתוב רשימת עובדות.
אל תכתוב מחמאות כלליות.
אל תכתוב אבחנות שיכולות להתאים כמעט לכל אדם.

בכל פסקה עליך לעשות אחד מהדברים:
- לזהות דפוס חוזר
- להסביר מתח פנימי
- לחבר בין פרט חיים לבין צורך זוגי
- להסביר למה העדפה זוגית מסוימת חשובה לאדם הזה
- להראות מה פחות מתאים לו/לה ולמה

מותר להזכיר פרטים מהשיחה, אבל רק אם הם מובילים לתובנה.
דוגמה לא טובה:
"את אוהבת לבשל ולטייל."
דוגמה טובה:
"האהבה שלך לבישול, יין וסרט בבית מצביעה על צורך בקשר שמרגיש חם, פשוט ולא מתאמץ — זוגיות שנבנית מהרגעים הקטנים, לא מדרמה או ריגוש תמידי."

חפש תמיד את ה"גם וגם":
- רכות לצד גבולות
- נתינה לצד צורך להתרחק מאנשים רעילים
- פתיחות לצד קווים אדומים
- חופש לצד רצון בביטחון
- עומק לצד צורך בפשטות

אל תנפח עומק שלא קיים. אם האדם פרקטי, קצר או פשוט בתשובות שלו — כתוב תובנות פשוטות ומדויקות, לא שפה טיפולית מוגזמת.
עוד שינוי קטן בקוד

החישוב של preCompletion לפי כמות כל ההודעות כולל גם הודעות assistant, ולכן הוא לא מאוד אמין:

const cogCount = allMessages.filter((m: any) => m.guide === "new_chat_cognitive").length;
const tasteCount = allMessages.filter((m: any) => m.guide === "new_chat_taste").length;

עדיף לספור רק הודעות משתמש:

const cogUserCount = allMessages.filter(
  (m: any) => m.guide === "new_chat_cognitive" && m.role === "user"
).length;

const tasteUserCount = allMessages.filter(
  (m: any) => m.guide === "new_chat_taste" && m.role === "user"
).length;

const preCompletion = cogUserCount < 3 || tasteUserCount < 3;

או אפילו עדיף לפי סטטוס שלבים אם יש לך בטבלה.

השינוי הכי חשוב בעיניי

להוסיף לפרומפט את המשפט הזה:

אם הפלט נשמע כמו טקסט שאפשר לתת לעוד משתמש/ת — הוא לא מספיק טוב. כתוב מחדש עד שהפסקאות יישענו על דפוסים ופרטים ייחודיים מהשיחה.

זה לב הבעיה. כרגע המודל כותב יפה, אבל לא מספיק “One”.

לגבי אלה: “לעולם אל תחזור על עובדות יבשות”
“לעולם אל תצטט מה נאמר בשיחה” זה כי הוא כל הזמן היה פשוט אומר דברים עובדתיים כמו - את אוהבת ים, את מעדיפה לצאת לבר מאשר למסיבה ופשוט חוזר על מה שהיא אמרה 

כן, לגמרי מבין למה שמת את זה. הבעיה היא לא הכוונה — היא הניסוח הקיצוני. כשהמודל רואה “לעולם אל תחזור על עובדות”, הוא לוקח את זה רחוק מדי ומפסיק להשתמש בפרטים בכלל. ואז הוא עובר ל”תובנות” מופשטות כמו אותנטיות, עומק, חום, פשטות, צמיחה — גם כשאין לזה מספיק אחיזה.

הייתי משאיר את הרעיון, אבל משנה את הכלל מאיסור על עובדות לאיסור על עובדות בלי פרשנות.

כלומר במקום:

- לעולם אל תחזור על עובדות יבשות
- לעולם אל תצטט מה נאמר בשיחה

הייתי כותב משהו כזה:

- אל תחזור על עובדות כפי שהמשתמש/ת אמר/ה אותן.
- אל תכתוב משפטים שהם רק תיאור עובדתי, בלי תובנה.
- מותר ואף רצוי להשתמש בפרטים קונקרטיים מהשיחה, אבל רק כשהם משרתים פרשנות עמוקה יותר.
- כל פרט שאתה מזכיר חייב לענות על השאלה: מה זה מלמד על האדם, על הצרכים שלו/ה או על ההתאמה הזוגית שלו/ה?

ואז לתת דוגמאות מאוד ברורות:

לא טוב:
"את אוהבת ים, מסעדות וסרטים בבית."

טוב:
"הבחירה שלך בערבים שקטים, אוכל טוב וסרט בבית מצביעה על צורך בקשר שמרגיש ביתי, חם ולא מתאמץ — כזה שנבנה מרגעים פשוטים ולא מחיפוש תמידי אחרי ריגוש."

לא טוב:
"את מעדיפה בר על פני מסיבה."

טוב:
"נראה שאת אוהבת חוויה חברתית שיש בה חיים ואווירה, אבל פחות נמשכת לעומס חברתי שאין בו אינטימיות או שליטה על הקצב."

לא טוב:
"את אוהבת לבשל."

טוב:
"הבישול אצלך הוא לא רק תחביב; הוא דרך לייצר חום, קרבה ותחושת בית בתוך קשר."

גם לגבי ציטוטים — הייתי מחדד. לא צריך ציטוטים ישירים, אבל כן אפשר להגיד “כשהתיאורים שלך חוזרים סביב...” בלי לצטט.

במקום:

- לעולם אל תצטט מה נאמר בשיחה

הייתי כותב:

- אל תשתמש בציטוטים ישירים מהשיחה ואל תכתוב "אמרת ש..." או "כשנשאלת...".
- במקום לצטט, עבד את התשובה לתובנה.
- כן אפשר להתייחס בעקיפין לדפוסים שחזרו בשיחה.

דוגמה:

לא טוב:
"כששאלו אותך מה הערב המושלם, אמרת שאת אוהבת לבשל ארוחה עם יין ולראות סרט."

טוב:
"הרבה מהבחירות שלך מצביעות על זוגיות שמרגישה כמו בית: אוכל טוב, ערב רגוע, שיחה נעימה ונוכחות של מישהי אחת שבאמת בוחרת בך."

הניסוח המדויק שהייתי מכניס לפרומפט:

## שימוש נכון בפרטים מהשיחה

אל תחזור על תשובות המשתמש/ת כמו שהן.
אל תכתוב רשימת עובדות, תחביבים או העדפות בלי פרשנות.
אל תשתמש בציטוטים ישירים, ואל תכתוב "אמרת ש..." או "כשנשאלת...".

כן השתמש בפרטים קונקרטיים מהשיחה כעוגנים לתובנה.
כל פרט שמופיע בפלט חייב להסביר משהו עמוק יותר:
- מה זה מלמד על הצורך הרגשי של המשתמש/ת
- מה זה מלמד על דפוס זוגי
- מה זה מלמד על סוג בן/בת הזוג שיתאים
- מה זה מלמד על מה שפחות יתאים

אם אין לך מה להסיק מפרט מסוים — אל תזכיר אותו.

דוגמאות:
לא טוב: "את אוהבת ים, ברים וסושי."
טוב: "נראה שאת צריכה קשר שיש בו חוויה, טעם ואווירה — לא בהכרח רעש גדול, אלא רגעים שמרגישים חיים, נעימים ומדויקים."

לא טוב: "את מעדיפה לקחת זמן להירגע בריב."
טוב: "בקונפליקט את צריכה רגע של ויסות לפני שיחה, ולכן בת זוג שתלחץ על פתרון מיידי עלולה להציף אותך יותר מאשר לקרב."

לא טוב: "את אוהבת לבשל."
טוב: "הבישול אצלך הוא דרך לייצר קרבה וביתיות, ולכן קשר טוב עבורך כנראה ייבנה גם דרך פעולות יומיומיות של דאגה ולא רק דרך שיחות גדולות."

כלומר: לא עובדות יבשות, כן עובדות מעוכלות.
זה בעיניי הניסוח שיפתור את שתי הבעיות: גם לא סיכום טכני של השיחה, וגם לא תובנות אווריריות מדי.
